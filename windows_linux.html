6.2 Installing and Configuring DNS Server

Add-DnsServerPrimaryZone `

    -NetworkID "192.168.1.0/24" `

Add-DnsServerResourceRecordA `

Add-DnsServerResourceRecordCName `
    -Name "ftp" `

Add-DnsServerResourceRecordMX `
    -Name "@" `
    -ZoneName "contoso.com" `

Add-DnsServerResourceRecordPtr `
    -Name "100" `
    -ZoneName "1.168.192.in-addr.arpa" `

6.3 DHCP (Dynamic Host Configuration Protocol)

Definition:
DHCP automatically assigns IP addresses and network configuration to clients.

DHCP Process (DORA):

Client          Server
  |──DISCOVER──→|  (Broadcast - looking for DHCP server)
  |←──OFFER────|  (Server offers IP address)
  |──REQUEST───→|  (Client requests offered IP)
  |←──ACK──────|  (Server confirms assignment)

DHCP Components:

  • Scope: Range of IP addresses
  • Exclusions: IPs excluded from distribution
  • Reservation: Fixed IP for specific device (based on MAC)
  • Lease Duration: How long client keeps the IP
  • Options: Gateway, DNS, WINS settings

6.4 Installing and Configuring DHCP

# Install DHCP Role
Install-WindowsFeature DHCP -IncludeManagementTools

# Authorize DHCP Server in AD
Add-DhcpServerInDC -DnsName "DC1.contoso.com" -IPAddress 192.168.1.10

# Create IPv4 Scope
Add-DhcpServerv4Scope `
    -Name "CorpNetwork" `
    -StartRange 192.168.1.100 `
    -EndRange 192.168.1.200 `
    -SubnetMask 255.255.255.0 `
    -Description "Corporate Network Scope" `
    -LeaseDuration 8.00:00:00

# Set Scope Options
Set-DhcpServerv4OptionValue `
    -ScopeId 192.168.1.0 `
    -Router 192.168.1.1 `
    -DnsServer 192.168.1.10 `
    -DnsDomain "contoso.com"

# Add Exclusion Range
Add-DhcpServerv4ExclusionRange `
    -ScopeId 192.168.1.0 `
    -StartRange 192.168.1.100 `
    -EndRange 192.168.1.110

# Create Reservation
Add-DhcpServerv4Reservation `
    -ScopeId 192.168.1.0 `
    -IPAddress 192.168.1.150 `
    -ClientId "AA-BB-CC-DD-EE-FF" `
    -Description "Printer Reservation"

# Activate/Deactivate Scope
Set-DhcpServerv4Scope -ScopeId 192.168.1.0 -State Active

# Configure DHCP Failover
Add-DhcpServerv4Failover `
    -Name "DHCP-Failover" `
    -ScopeId 192.168.1.0 `
    -PartnerServer "DC2.contoso.com" `
    -Mode HotStandby `
    -ServerRole Active

# View leases
Get-DhcpServerv4Lease -ScopeId 192.168.1.0

# Configure IPv6 Scope
Add-DhcpServerv6Scope `
    -Name "IPv6Scope" `
    -Prefix "2001:db8::" `
    -State Active

8.2 Group Policy Objects (GPO)

Definition:
GPO is a collection of settings that control the working environment for users and computers in Active Directory.

GPO Processing Order (LSDOU):

1. Local Policy
2. Site Policy
3. Domain Policy
4. OU Policy (parent OU first, then child OU)

Last policy applied wins (unless blocked or enforced)

GPO Scope:

  • Linked: Linked to Site, Domain, or OU
  • Filtered: WMI filters and Security filters
  • Enforced: Cannot be blocked by child containers
  • Block Inheritance: Child OU ignores parent GPOs

1.4 File Permissions

Permission Structure:

-rwxr-xr--  1  jsmith  users  4096  Jan 1 12:00  filename
│└┬┘└┬┘└┬┘           
│ │  │  └── Other permissions (r--)
│ │  └───── Group permissions (r-x)
│ └──────── Owner permissions (rwx)
└────────── File type (- file, d dir, l link)

r = read    (4)
w = write   (2)
x = execute (1)
- = no permission (0)

chmod - Change Permissions:

# Symbolic mode
chmod u+x file              # Add execute for user
chmod g-w file              # Remove write for group
chmod o=r file              # Set other to read only
chmod a+x file              # Add execute for all
chmod u+x,g+r file          # Multiple changes

# Numeric (octal) mode
chmod 755 file              # rwxr-xr-x
chmod 644 file              # rw-r--r--
chmod 700 file              # rwx------
chmod 777 file              # rwxrwxrwx
chmod 600 file              # rw-------

# Recursive
chmod -R 755 directory/

# Common permissions
# 755 - Directories, executables
# 644 - Regular files
# 600 - Private files (SSH keys)
# 700 - Private directories

Special Permissions:

# SUID (Set User ID) - Run as file owner
chmod u+s /usr/bin/passwd   # Or chmod 4755
# Shown as 's' in user execute position

# SGID (Set Group ID) - Run as group / inherit group
chmod g+s directory/        # Or chmod 2755
# Files in directory inherit group

# Sticky Bit - Only owner can delete
chmod +t /tmp               # Or chmod 1755
# Shown as 't' in other execute position

chown - Change Ownership:

chown username file              # Change owner
chown username:group file        # Change owner and group
chown :group file                # Change group only
chown -R username:group dir/     # Recursive

#!/bin/bash
# System Health Check Script

LOG="/var/log/health_check.log"
EMAIL="admin@contoso.com"
THRESHOLD=80

log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a $LOG
}

check_cpu() {
    CPU_USAGE=$(top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'%' -f1)
    log "CPU Usage: $CPU_USAGE%"
    if (( $(echo "$CPU_USAGE > $THRESHOLD" | bc -l) )); then
        log "WARNING: High CPU usage: $CPU_USAGE%"
        return 1
    fi
    return 0
}

check_memory() {
    MEM_USAGE=$(free | grep Mem | awk '{printf "%.0f", $3/$2 * 100}')
    log "Memory Usage: $MEM_USAGE%"
    if [ $MEM_USAGE -gt $THRESHOLD ]; then
        log "WARNING: High memory usage: $MEM_USAGE%"
        return 1
    fi
    return 0
}

check_disk() {
    while IFS= read -r line; do
        USAGE=$(echo $line | awk '{print $5}' | cut -d'%' -f1)
        MOUNT=$(echo $line | awk '{print $6}')
        log "Disk $MOUNT: $USAGE%"
        if [ $USAGE -gt $THRESHOLD ]; then
            log "WARNING: Disk $MOUNT is $USAGE% full!"
        fi
    done < <(df -h | tail -n +2)
}

check_services() {
    SERVICES=("httpd" "sshd" "named" "dhcpd")
    for svc in "${SERVICES[@]}"; do
        if systemctl is-active --quiet $svc; then
            log "Service $svc: RUNNING"
        else
            log "ALERT: Service $svc is DOWN!"
            systemctl start $svc && log "Service $svc restarted"
        fi
    done
}

# Main
log "=== Health Check Started ==="
check_cpu
check_memory
check_disk
check_services
log "=== Health Check Completed ==="

21.8 Log Monitoring Script

#!/bin/bash
# Log Monitoring and Alerting Script

LOGFILE="/var/log/messages"
ALERT_LOG="/var/log/alerts.log"
KEYWORDS=("error" "critical" "failed" "denied" "attack")
LAST_CHECK="/tmp/last_check_position"

monitor_logs() {
    # Get file size
    CURRENT_SIZE=$(stat -c %s "$LOGFILE")
    
    if [ -f "$LAST_CHECK" ]; then
        LAST_POS=$(cat "$LAST_CHECK")
    else
        LAST_POS=0
    fi
    
    # Read new lines
    if [ $CURRENT_SIZE -gt $LAST_POS ]; then
        NEW_CONTENT=$(tail -c +$LAST_POS "$LOGFILE")
        
        for keyword in "${KEYWORDS[@]}"; do
            MATCHES=$(echo "$NEW_CONTENT" | grep -i "$keyword")
            if [ -n "$MATCHES" ]; then
                echo "[$(date)] ALERT: $keyword found in logs:" >> "$ALERT_LOG"
                echo "$MATCHES" >> "$ALERT_LOG"
                # Send email alert
                echo "$MATCHES" | mail -s "Log Alert: $keyword" admin@company.com
            fi
        done
    fi
    
    echo $CURRENT_SIZE > "$LAST_CHECK"
}

# Run continuously
while true; do
    monitor_logs
    sleep 60   # Check every minute
done
